851: // Обновляет отображение количества "любимых персонажей" в блоке статистики
852: function updateFavoriteCharactersStat(count) {
853:     const statCards = document.querySelectorAll('.stat-card');
854:     statCards.forEach(card => {
855:         const labelEl = card.querySelector('.stat-label');
856:         const numberEl = card.querySelector('.stat-number');
857:         if (!labelEl || !numberEl) return;
858: 
859:         if (labelEl.textContent.trim().includes('любимых персонажей')) {
860:             numberEl.textContent = count;
861:             // Если анимация счетчика уже проигралась, проигрываем заново с новым значением
862:             if (card.dataset.animated === 'true') {
863:                 animateCounter(numberEl, String(count), 1500);
864:             }
865:         }
866:     });
867: }

869: // Инициализирует динамическую загрузку статистики на страницах без каталога персонажей
870: async function initDynamicStats() {
871:     const statsGrid = document.querySelector('.stats-grid');
872:     if (!statsGrid) return;
873:     
874:     // Если на странице нет сетки персонажей, но есть счетчики, загружаем данные для синхронизации
875:     if (!document.getElementById('characters-grid')) {
876:         try {
877:             const response = await fetch('data/characters-data.json');
878:             if (response.ok) {
879:                 const data = await response.json();
880:                 if (data && Array.isArray(data.characters)) {
881:                     updateFavoriteCharactersStat(data.characters.length);
882:                 }
883:             }
884:         } catch (error) {
885:             console.error('Error loading dynamic statistics:', error);
886:         }
887:     }
888: }
